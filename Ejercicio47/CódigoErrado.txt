class Producto {
    String nombre;
    int stock;

    void mostrarProducto() {
        System.out.println(nombre + " - stock: " + stock);
    }
}

public class Main {
    public static void main(String[] args) {
        Producto p;               // solo se declara la referencia, pero no se instancia
        p.nombre = "Teclado";     // error en compilacion
        p.mostrarProducto();
    }
}
